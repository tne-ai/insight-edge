df = session.get_object("handbags-shopping_list-no_run-1_1-with_periods_cleaned.csv")

# Calculate Gross Margin for each row
df['gross_margin'] = (df['aur'] - df['standard_cost_price']) * df['sales']

# Group by product_subgroup_desc to calculate Total Margin and Total Sales Revenue
grouped = df.groupby('product_subgroup_desc').agg(
    total_margin=('gross_margin', 'sum'),
    total_sales_revenue=('sales v', 'sum')
).reset_index()

# Calculate Gross Margin Percentage
grouped['gross_margin_percentage'] = (grouped['total_margin'] / grouped['total_sales_revenue']) * 100

# Filter for subclasses with negative Total Margin
result = grouped[grouped['total_margin'] < 0].sort_values(by='total_margin', ascending=True)
