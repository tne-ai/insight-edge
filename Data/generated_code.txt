import pandas as pd
from tne.TNE import TNE

df = session.get_object("handbags-shopping_list-no_run-1_1-with_periods_cleaned.csv")

# Filter data for December
df['period'] = pd.to_numeric(df['period'], errors='coerce')
december_data = df[df['period'] % 100 == 12]

# Group by 'product_subgroup_desc' (subclass) and sum the sales
top_subclasses = (
    december_data.groupby('product_subgroup_desc', as_index=False)['sales']
    .sum()
    .sort_values(by='sales', ascending=False)
    .head(12)
)

# Store the result
result = top_subclasses
