df = session.get_object("handbags-shopping_list-no_run-1_1-with_periods_cleaned.csv")

# Filter data for tote bags
tote_bags = df[df['product_subgroup_desc'] == 'Tote bag']

# Calculate total sales for tote bags
total_sales_tote_bags = tote_bags['sales'].sum()

# Group by price metric and calculate sales contribution
contribution = (
    tote_bags.groupby('price_metric')['sales']
    .sum()
    .reset_index()
    .rename(columns={'sales': 'price_metric_sales'})
)
contribution['contribution_percentage'] = (contribution['price_metric_sales'] / total_sales_tote_bags) * 100

# Sort by contribution percentage in descending order
contribution = contribution.sort_values(by='contribution_percentage', ascending=False).reset_index(drop=True)
