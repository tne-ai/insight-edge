# Convert 'period' column to datetime for sorting
df['period'] = pd.to_datetime(df['period'], format='%Y%m')

# Get the last available period
last_period = df['period'].max()

# Filter data for the last period
last_period_data = df[df['period'] == last_period]

# Calculate Sell-Through Rate (STR)
last_period_data['sell_through_rate'] = (last_period_data['sales'] / 
                                         (last_period_data['sales'] + last_period_data['end_inv'])) * 100

# Select the 10 products with the highest ending inventory
high_inventory_stock = last_period_data.nlargest(10, 'end_inv')
