import pandas as pd
from tne.TNE import TNE

# Initialize session and get data
session = TNE(uid=UID, bucket_name=BUCKET, project=PROJECT, version=VERSION)
df = session.get_object("womens_rtw-shopping_list-442-1_1-with_periods-with_sizes_cleaned.csv")

# Filter for Women Shorts in the product_subgroup_desc column
shorts_df = df[df['product_subgroup_desc'] == 'WOMENS SHORTS']

# Convert product_launch_date to datetime for filtering
shorts_df['product_launch_date'] = pd.to_datetime(shorts_df['product_launch_date'], format='%d/%m/%Y', errors='coerce')

# Filter for products launched in February
february_shorts = shorts_df[shorts_df['product_launch_date'].dt.month == 2]

# Group by SKU (sku_id) and sum sales to rank by sales performance
top_skus = february_shorts.groupby(['sku_id', 'style_color_description', 'size_code'], as_index=False)['sales'].sum()

# Sort by sales in descending order and select the top 10 SKUs
top_skus = top_skus.sort_values(by='sales', ascending=False).head(10)

# Select relevant columns for the result
result = top_skus[['sku_id', 'style_color_description', 'size_code', 'sales']]
